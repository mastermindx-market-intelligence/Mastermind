def sources():
    assert util.find_spec("portfolio.decision_snapshot_sources") is not None, "missing fixed source capture"
    return import_module("portfolio.decision_snapshot_sources")


def write_json(path, value, *, stamp="2026-09-15T19:59:00Z"):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value))
    epoch = datetime.fromisoformat(stamp.replace("Z", "+00:00")).timestamp()
    os.utime(path, (epoch, epoch))

def capture_all(s):
    return s.capture_all("autonomous", decision_cutoff="2026-09-15T20:00:00Z", recorded_at="2026-09-15T20:01:00Z")


def by_id(result):
    return {r["source_id"]: r for r in result["sources"]}


def test_capture_is_read_only_and_describes_all_domains(roots):
    s, repo, macro = roots
    write_json(repo / "data/portfolios/autonomous/account.json", {"cash": 800000, "positions": {"AAPL": {"shares": 100, "avg_cost": 180}}})
    write_json(macro / "site/riskdata/risk_envelope.json", {"schema": "mastermind.risk_envelope/v1", "generated_at": "2026-09-15T19:55:00Z", "as_of": "2026-09-15", "data_state": "FRESH", "capital_policy": {"posture": "SELECTIVE"}})
    before = {p: p.read_bytes() for p in repo.rglob("*") if p.is_file()}
    result = capture_all(s); receipts = by_id(result)
    assert receipts["book.account"]["status"] == "AVAILABLE"
    assert receipts["macro.risk_envelope"]["known_at"] == "2026-09-15T19:55:00Z"
    assert receipts["book.pending_target"]["status"] == "ABSENT_OPTIONAL"
    assert len(result["sections"]) == 12
    assert {p: p.read_bytes() for p in repo.rglob("*") if p.is_file()} == before
    for receipt in result["sources"]: s.c.validate_source_receipt(receipt)
    for section in result["sections"].values(): s.c.validate_section(section)
