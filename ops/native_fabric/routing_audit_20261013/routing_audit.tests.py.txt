def broker(tmp_path, accounts=None):
    import yaml
    accounts = accounts or [{"name": "synthetic", "per_account_cap": 1}]
    source = tmp_path / "accounts.yaml"
    source.write_text(yaml.safe_dump({"providers": {
        "minimax": {"enabled": True, "accounts": accounts}}}))
    return lease_broker.Broker(db_path=str(tmp_path / "leases.sqlite"),
        lock_path=str(tmp_path / "leases.lock"), accounts_path=str(source),
        now_fn=lambda: 2_000_000_000.0, check_pid=False, external_gates=False,
        grok_marker=str(tmp_path / "absent"), status_json=str(tmp_path / "status.json"),
        hosts_path=str(tmp_path / "hosts.json"), this_host="synthetic")


def require_logical_class_api():
    assert "job_class" in inspect.signature(lease_broker.Broker.acquire).parameters, (
        "remote transport labels currently erase logical job priority; add a separate optional job_class")


def test_remote_heal_overtakes_older_mechanical_without_losing_transport_identity(tmp_path):
    require_logical_class_api()
    br = broker(tmp_path)
    first = br.acquire("holder", "minimax", task_class="remote-minimax", job_class="mechanical")
    assert first.ok
    older = br.acquire("older", "minimax", task_class="remote-minimax", job_class="mechanical")
    urgent = br.acquire("urgent", "minimax", task_class="remote-minimax", job_class="heal")
    assert not older.ok and not urgent.ok
    assert br.release(first.lease_id, "holder")
    result = br.acquire("urgent", "minimax", task_class="remote-minimax", job_class="heal")
    assert result.ok, result
    with br._conn() as conn:
        row = conn.execute("SELECT task_class FROM leases WHERE id=?", (result.lease_id,)).fetchone()
        assert row["task_class"] == "remote-minimax"
        waiting = conn.execute("SELECT priority FROM waiters WHERE orchestrator_id='older'").fetchone()
        assert waiting["priority"] == 3

@pytest.mark.parametrize("job_class,priority", [
    ("heal", 0), ("build", 1), ("review", 2), ("mechanical", 3),
    ("unexpected", 4), ("", 4), ("  Review  ", 2),
])
def test_remote_waiter_priority_uses_logical_class(tmp_path, job_class, priority):
    require_logical_class_api()
    br = broker(tmp_path)
    assert br.acquire("holder", "minimax").ok
    assert not br.acquire("waiting", "minimax", task_class="remote-minimax", job_class=job_class).ok
    with br._conn() as conn:
        row = conn.execute("SELECT priority, task_class FROM waiters WHERE orchestrator_id='waiting'").fetchone()
        assert row["priority"] == priority
        assert row["task_class"] == "remote-minimax"


def test_glm_pairing_refusal_is_not_bypassed_by_logical_class(tmp_path, monkeypatch):
    require_logical_class_api()
    br = broker(tmp_path)
    seen = []
    def unavailable(accounts, **kwargs):
        seen.append(kwargs)
        return [], ["not_qualified"]
    monkeypatch.setitem(sys.modules, "glm_host_account_source",
                        types.SimpleNamespace(filter_observed_accounts=unavailable))
    answer = br.acquire("root", "glm", task_class="remote-glm", job_class="heal",
                        model="glm-5.3", host="synthetic")
    assert not answer.ok and answer.reason == "host_account_unavailable"
    assert seen and seen[0]["model"] == "glm-5.3"
    with br._conn() as conn:
        assert conn.execute("SELECT COUNT(*) FROM leases").fetchone()[0] == 0
        assert conn.execute("SELECT COUNT(*) FROM waiters").fetchone()[0] == 0


def test_all_remote_acquire_paths_forward_both_class_dimensions():
    source = (EXT / "remote_sub.sh").read_text()
    calls = [line for line in source.splitlines()
             if ('broker acquire' in line or 'show_args "<broker>" acquire' in line)]
    assert len(calls) == 3
    for line in calls:
        assert '--task-class "remote-$MODE"' in line
        assert '--job-class "${POOL_TASK_CLASS:-}"' in line, line
