def iso(t):
    return datetime.fromtimestamp(t, timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")


def sample_account(now, *, weekly_reset=6 * 3600, five_reset=3600,
                   weekly_pct=70, five_pct=75, age=25, source="PROVIDER_AUTHORITATIVE",
                   provider="glm", observed_age=None, duplicate=False):
    stamp = iso(now - (age if observed_age is None else observed_age))
    week = dict(provider=provider, window="weekly", used=30, cap=100,
                remaining_pct=weekly_pct, reset_at=iso(now + weekly_reset),
                source=source, observed_at=stamp, confidence=0.98, detail="test-only")
    five = dict(provider=provider, window="5h", used=25, cap=100,
                remaining_pct=five_pct, reset_at=iso(now + five_reset),
                source=source, observed_at=stamp, confidence=0.98, detail="test-only")
    return dict(refreshed_at=now-age, states=[week, five] + ([dict(week)] if duplicate else []))

def test_urgency_selects_matching_account_in_actual_atomic_claim(case):
    br, now, data, save, acquire, _ = case
    r = acquire("first")
    assert r.ok and r.account == "zeta_near_reset"
    assert r.lease_id
    with br._conn() as conn:

    assert (record["account"],record["model"],record["host"])==("zeta_near_reset","glm-5.3-flash","ubuntu0")

@pytest.mark.parametrize("mutation", [
    lambda d,n: d[n].update(refreshed_at=FIXED-1200),
    lambda d,n: d[n]["states"][0].update(observed_at=iso(FIXED-1200)),
    lambda d,n: d[n]["states"][0].update(reset_at=iso(FIXED-5)),
    lambda d,n: d[n]["states"][0].update(reset_at=iso(FIXED+40*3600)),
    lambda d,n: d[n]["states"][0].update(remaining_pct=10),
    lambda d,n: d[n]["states"][0].update(remaining_pct=None),
    lambda d,n: d[n]["states"][0].update(remaining_pct=True),
    lambda d,n: d[n]["states"][0].update(remaining_pct=float("nan")),
    lambda d,n: d[n]["states"][0].update(source="LOCAL_RECONSTRUCTION"),
    lambda d,n: d[n]["states"][0].update(provider="minimax"),
    lambda d,n: d[n]["states"][1].update(remaining_pct=0),
    lambda d,n: d[n]["states"][1].update(source="LOCAL_RECONSTRUCTION"),
    lambda d,n: d[n]["states"][1].update(observed_at=iso(FIXED-20)),
    lambda d,n: d[n]["states"][1].update(reset_at=iso(FIXED-2)),
    lambda d,n: d[n]["states"].pop(),
    lambda d,n: d[n]["states"].append(dict(d[n]["states"][0])),
])
def test_invalid_or_stale_window_never_promotes_account(case,mutation):
    br, now, data, save, acquire, _=case
    mutation(data["glm"]["accounts"], "zeta_near_reset")
    save()
    # Neither account is urgent; original stable alphabetical tie-break.
    assert acquire("invalid").account=="alpha_far_reset"

def test_freshness_reader_errors_fail_neutral_without_new_authority(case,monkeypatch):

def test_model_cap_is_hard_even_for_urgent_account(case):
    br, now, data, save, acquire, _=case
    # Bound model-specific cap on urgent account to 0, and preserved normal
    # account capacity can still admit the other qualified account.
    data["glm"]["accounts"]["zeta_near_reset"]["states"][1]["remaining_pct"]=85
    save()
    source_provider=br.provider
    def changed_provider(pool):
        provider=source_provider(pool)
        if provider["_name"]=="glm":
            for row in provider["accounts"]:
                if row["name"]=="zeta_near_reset":
                    row["per_model_caps"]={"glm-5.3-flash":0}
        return provider
    br.provider=changed_provider
    assert acquire("model-limited").account=="alpha_far_reset"


def test_concurrent_claims_not_double_overcap(case):
    br, now, data, save, acquire, _=case
    barrier=threading.Barrier(6)
    results=[]
    mutex=threading.Lock()
    def worker(i):
        barrier.wait()
        result=acquire("parallel-"+str(i))
        with mutex:
            results.append(result)
    threads=[threading.Thread(target=worker,args=(i,)) for i in range(6)]
    for t in threads:t.start()
    for t in threads:t.join(timeout=15)
    assert len(results)==6
    granted=[r for r in results if r.ok]
    assert len(granted)==4
    assert sum(r.account=="zeta_near_reset" for r in granted)==2
    assert sum(r.account=="alpha_far_reset" for r in granted)==2
    assert len({r.lease_id for r in granted})==4

def test_cache_symlink_swap_between_stat_and_open_cannot_promote(case, monkeypatch, tmp_path):
    """Adversarial path swap after lstat cannot change the opened quota file."""
    br, now, data, save, acquire, cache = case
    target = tmp_path / "other.json"
    target.write_text(json.dumps(data))
    original_open = lease_broker.os.open
    changed = [False]

    def swap(path, *args, **kwargs):
        if not changed[0] and str(path) == str(cache):
            cache.unlink()
            cache.symlink_to(target)
            changed[0] = True
        return original_open(path, *args, **kwargs)

    monkeypatch.setattr(lease_broker.os, "open", swap)
    assert br._fresh_glm_expiring_accounts(set(ACCOUNT_NAMES)) == {}
    assert changed == [True]


def test_cache_modified_while_open_is_not_trusted(case, monkeypatch):
    """A simultaneous provider cache replace/in-place edit cannot combine epochs."""
    br, now, data, save, acquire, cache = case
    original_read = lease_broker.os.read
    touched = [False]

    def mutate(fd, n):
        chunk = original_read(fd, n)
        if not touched[0]:
            touched[0] = True
            cache.write_text('{"glm":{"accounts":{}}}\n' + ' ' * 120)
        return chunk

    monkeypatch.setattr(lease_broker.os, "read", mutate)
    assert br._fresh_glm_expiring_accounts(set(ACCOUNT_NAMES)) == {}
    assert touched == [True]

def test_host_qualified_account_subset_precedes_expiry_preference(case, monkeypatch):
    """The current host+model qualification owner is never overridden by expiry."""
    import glm_host_account_source
    br, now, data, save, acquire, cache = case
    seen = []

    def compatible(rows, *, host, model, now, preferred, hosts_path):
        seen.append((host, model, preferred))
        return ([row for row in rows if row["name"] == "alpha_far_reset"], ())

    monkeypatch.setattr(glm_host_account_source, "filter_observed_accounts", compatible)
    actual = br.acquire(
        orchestrator_id="host-bound-1", pool="glm", account=None,
        model="glm-5.3-flash", task_class="remote-glm", host="ubuntu0",
        pid=os.getpid(), ttl=120, wait=0, poll=.01,
    )
    assert actual.ok and actual.account == "alpha_far_reset"
    assert seen == [("ubuntu0", "glm-5.3-flash", None)]
    with br._conn() as conn:
        row = conn.execute("SELECT account, model, host FROM leases WHERE id=?",
                           (actual.lease_id,)).fetchone()
    assert (row["account"], row["model"], row["host"]) == (
        "alpha_far_reset", "glm-5.3-flash", "ubuntu0",
    )


def test_explicit_account_outside_host_qualified_subset_stays_refused(case, monkeypatch):
    import glm_host_account_source
    br, now, data, save, acquire, cache = case

    def compatible(rows, *, host, model, now, preferred, hosts_path):
        allowed = ([row for row in rows if row["name"] == "alpha_far_reset"]
                   if preferred in (None, "alpha_far_reset") else [])
        return allowed, ()

    monkeypatch.setattr(glm_host_account_source, "filter_observed_accounts", compatible)
    result = br.acquire(
        orchestrator_id="host-refused-1", pool="glm", account="zeta_near_reset",
        model="glm-5.3-flash", task_class="remote-glm", host="ubuntu0",
        pid=os.getpid(), ttl=120, wait=0, poll=.01,
    )
    assert not result.ok and result.reason == "host_account_unavailable"
    with br._conn() as conn:
        rows = conn.execute("SELECT COUNT(*) AS n FROM leases").fetchone()
    assert rows["n"] == 0
